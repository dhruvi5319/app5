**Test cases to cover:**

```
addTask — valid input:
  - addTask('Buy milk') → returns task object with { id, name: 'Buy milk', completed: false, createdAt }
  - addTask('  Trimmed  ') → returns task with name: 'Trimmed' (trimmed)
  - addTask('') → returns null (invalid), does not push to tasks

toggleTask — known id:
  - given task with completed=false, toggleTask(task.id) → task.completed is now true
  - given task with completed=true, toggleTask(task.id) → task.completed is now false (round-trip)

toggleTask — unknown id:
  - toggleTask('nonexistent-id') → no throw, state unchanged

deleteTask — known id:
  - addTask('To be deleted'), deleteTask(task.id) → task no longer findable by id

deleteTask — unknown id:
  - deleteTask('nonexistent-id') → no throw, task count unchanged
```

**Resetting state between tests:** The `tasks` array in app.js is module-level state. Since Node.js caches require(), all tests share the same `tasks` array instance. Reset between tests by calling `deleteTask` on each created task, or by tracking created tasks and cleaning up in afterEach — but `node:test` has `beforeEach`/`afterEach` hooks:

```javascript
const { beforeEach } = require('node:test');
// After each test, clear tasks by re-requiring or tracking IDs
```

Simplest reliable approach: track task ids created in each test and delete them in cleanup, OR use `test.before`/`after` to reset — but module cache means we can't re-require. Best approach: in each test, create the task needed AND immediately delete it as part of cleanup in a finally block or by using the test callback's context.

Alternative: export a `_resetTasksForTesting` helper from app.js that sets `tasks = []`. Add it to module.exports (test-only):

```javascript
// In module.exports block, after existing exports:
_resetTasksForTesting: () => { tasks = []; },
```

Then call `_resetTasksForTesting()` at the start of each test (or in beforeEach).

Use this approach — it's clean, explicit, and test-only (browser ignores module.exports).

**Complete test structure example:**

```javascript
'use strict';
const { test, beforeEach } = require('node:test');
const assert = require('node:assert/strict');

// [localStorage mock here]
// [document mock here]

const {
  addTask, toggleTask, deleteTask, _resetTasksForTesting
} = require('../app.js');

beforeEach(() => { _resetTasksForTesting(); });

test('addTask with valid name returns task object', () => {
  const task = addTask('Buy milk');
  assert.ok(task);
  assert.equal(task.name, 'Buy milk');
  assert.equal(task.completed, false);
  assert.ok(task.id);
  assert.ok(task.createdAt);
});

test('addTask with trimmed name stores trimmed value', () => {
  const task = addTask('  Trimmed  ');
  assert.equal(task.name, 'Trimmed');
});

test('addTask with empty string returns null', () => {
  const result = addTask('');
  assert.equal(result, null);
});

test('toggleTask flips completed from false to true', () => {
  const task = addTask('Toggle me');
  assert.equal(task.completed, false);
  toggleTask(task.id);
  assert.equal(task.completed, true);
});

test('toggleTask flips completed from true back to false', () => {
  const task = addTask('Toggle round-trip');
  toggleTask(task.id);   // → true
  toggleTask(task.id);   // → false
  assert.equal(task.completed, false);
});

test('toggleTask with unknown id does not throw', () => {
  assert.doesNotThrow(() => toggleTask('nonexistent-id'));
});

test('deleteTask removes task from list', () => {
  const task = addTask('Delete me');
  deleteTask(task.id);
  // After delete, adding and toggling same id has no effect
  toggleTask(task.id);     // should warn but not throw
  assert.doesNotThrow(() => toggleTask(task.id));
});

test('deleteTask with unknown id does not throw', () => {
  assert.doesNotThrow(() => deleteTask('nonexistent-id'));
});
```
